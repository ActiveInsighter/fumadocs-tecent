export const versionInfo = { next: 16, react: 19, tailwind: 4 };
export const surfaces = [
  ['Background', 'bg-background text-foreground'],
  ['Card', 'bg-card text-card-foreground'],
  ['Muted', 'bg-muted text-muted-foreground'],
  ['Accent', 'bg-accent text-accent-foreground'],
];

export function StatusBadge({ children }) {
  return (
    <span className="inline-flex items-center rounded-full border bg-fd-accent px-2.5 py-1 text-sm font-medium text-fd-accent-foreground">
      {children}
    </span>
  );
}

<Callout title="MDX 是 Markdown 的超集" type="info">
  除了普通 Markdown，MDX 还允许导出变量、执行安全的构建期表达式、定义 React 组件，并在正文中混合 JSX。
</Callout>

## JavaScript 表达式

当前框架名称：**{frameworkName}**。

版本组合：Next.js **{versionInfo.next}**、React **{versionInfo.react}**、Tailwind CSS **{versionInfo.tailwind}**。

字符串表达式：`{'MDX'.toLowerCase()}` 的运行结果是 **{'MDX'.toLowerCase()}**。

简单计算：`21 × 2 =` **{21 * 2}**。

## 自定义组件

下面的 Badge 直接定义在当前 MDX 文件中：

<StatusBadge>Theme ready</StatusBadge>{' '}
<StatusBadge>MDX enabled</StatusBadge>{' '}
<StatusBadge>Build-time rendering</StatusBadge>

```mdx title="自定义组件语法"
export function StatusBadge({ children }) {
  return <span className="rounded-full border px-2.5 py-1">{children}</span>;
}

<StatusBadge>Theme ready</StatusBadge>
```

## JSX 注释

{/* 这条注释不会渲染到最终页面。 */}

JSX 注释使用花括号包裹，适合给文档作者留下维护说明，而不会出现在读者看到的正文中。
